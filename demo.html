기술 예시

Lab Track 참가 승인을 받은 창작자를 위한 예시입니다. 받은 PNG 와 아틀라스 JSON 만으로 애니메이션을 원본 속도로 재생하는 방법을 보여 줍니다. 서버나 빌드 도구 없이 브라우저에서 그대로 돌아가는 코드입니다.

아틀라스로 애니메이션 재생하기

아틀라스 JSON 의 frames 는 시트 안 프레임 위치, animations 는 재생 순서(frames)와 프레임마다 머무는 시간(frame_ms)입니다. 프레임마다 크기가 달라도 center(기준점)를 한 자리에 맞춰 그리면 흔들리지 않습니다.

const atlas = await (await fetch(base + id + '.json')).json();
const sheet = new Image();
sheet.src = base + atlas.sheet.file;
await sheet.decode();

const anim = atlas.animations[0];        // 재생 순서와 지연
let step = 0;
function draw() {
  const f = atlas.frames[anim.frames[step]];
  ctx.clearRect(0, 0, cv.width, cv.height);
  ctx.imageSmoothingEnabled = false;     // 픽셀 그림은 보간하지 않는다
  // 기준점(center)을 화면의 한 점(ax, ay)에 맞춘다
  ctx.drawImage(sheet, f.x, f.y, f.w, f.h,
    (ax - f.center.x) * scale, (ay - f.center.y) * scale, f.w * scale, f.h * scale);
  setTimeout(draw, anim.frame_ms[step]); // 원본 지연대로
  step = (step + 1) % anim.frames.length;
}
draw();
